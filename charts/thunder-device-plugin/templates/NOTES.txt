Components:
  operator  {{ if .Values.operator.enabled }}enabled  - publishes ResourceSlice inventory{{ else }}DISABLED - no inventory will be published{{ end }}
  daemon    DaemonSet on nodes labelled {{ .Values.nodeLabelKeys.thunderNode }}=true
  driver    {{ .Values.dra.driverName }}
  classes   {{ if .Values.operator.extendedResourcePrefix }}one per GPU type, generated from Thunder inventory{{ else }}DISABLED - create DeviceClasses yourself{{ end }}

1. Wait for both components to become ready:

     kubectl -n {{ .Values.namespace.name }} rollout status daemonset/{{ include "thunder-device-plugin.fullname" . }}
{{- if .Values.operator.enabled }}
     kubectl -n {{ .Values.namespace.name }} rollout status deployment/{{ include "thunder-device-plugin.fullname" . }}-operator
{{- end }}

2. Confirm the operator published GPU inventory:

     kubectl get deviceclasses -l app.kubernetes.io/name=thunder-dra-driver \
       -o custom-columns=CLASS:.metadata.name,RESOURCE:.spec.extendedResourceName

   Or run the packaged smoke test:

     helm test {{ .Release.Name }} --namespace {{ .Values.namespace.name }}

3. Request GPUs. Every request names a GPU model; there is no "any GPU" form.
{{- if .Values.operator.extendedResourcePrefix }}

   As an extended resource (needs Kubernetes 1.36+):

     resources:
       limits:
         {{ .Values.operator.extendedResourcePrefix }}<gpu-type>: 2
{{- end }}

   Or as a ResourceClaim, which works on any supported cluster:

     apiVersion: resource.k8s.io/v1
     kind: ResourceClaimTemplate
     metadata:
       name: thunder-pod-gpu
     spec:
       spec:
         devices:
           requests:
             - name: gpu
               exactly:
                 deviceClassName: {{ .Values.operator.deviceClassPrefix }}<gpu-type>
                 allocationMode: ExactCount
                 count: 2          # one device per GPU
